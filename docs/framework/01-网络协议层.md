# 01 · 网络协议层（shared/NetworkProtocol.lua + ErrorCodes.lua）

> 网络协议层是前后端共用的**信封与校验**基础。它定义了"客户端 → 服务端"的命令如何打包、服务端响应如何返回、所有入参如何做 Schema 校验。这是整个框架最底层、最可复用的部分。

文件：
- `scripts/shared/NetworkProtocol.lua`（469 行）
- `scripts/shared/ErrorCodes.lua`（125 行）

---

## 1. 设计目标

1. **传输中立**：不直接绑定 UrhoX 网络对象，`VariantMap` 封装被隔离在 `ToVariantMap`，便于单元测试与替换宿主。
2. **一次校验，两端复用**：`ValidateValue`/`ValidatePayload` 是通用 Schema 校验器，服务端校验命令入参，客户端校验快照/增量。
3. **大小与格式安全**：所有字符串有长度上限，payload 有字节上限，command/requestId 有严格字符白名单。

---

## 2. 协议常量（信封格式）

```lua
local NetworkProtocol = {}

NetworkProtocol.VERSION = 1
NetworkProtocol.MAX_PAYLOAD_BYTES = 16 * 1024          -- 客户端→服务端 payload 上限
NetworkProtocol.MAX_SERVER_PAYLOAD_BYTES = 60 * 1024   -- 服务端→客户端 payload 上限
NetworkProtocol.SNAPSHOT_CHUNK_TARGET_BYTES = 56 * 1024
NetworkProtocol.SNAPSHOT_ITEMS_PER_CHUNK = 50
NetworkProtocol.MAX_REQUEST_ID_LENGTH = 96
NetworkProtocol.MAX_COMMAND_LENGTH = 64

-- 远程事件名（UrhoX RegisterRemoteEvent 用）
NetworkProtocol.EVENTS = {
    CLIENT_READY = "AbyssClientReady",  -- 客户端握手完成，可收快照
    COMMAND      = "AbyssCommand",      -- 客户端 → 服务端：命令
    RESPONSE     = "AbyssResponse",     -- 服务端 → 客户端：命令响应
    SNAPSHOT     = "AbyssSnapshot",     -- 服务端 → 客户端：全量状态（分片）
    DELTA        = "AbyssDelta",        -- 服务端 → 客户端：增量
    STATUS       = "AbyssStatus",       -- 服务端 → 客户端：状态机消息
}

-- VariantMap 字段名
NetworkProtocol.FIELDS = {
    VERSION     = "Version",
    REQUEST_ID  = "RequestId",
    COMMAND     = "Command",
    PAYLOAD_JSON = "PayloadJson",
}

-- 谁注册哪些远程事件（收发方向）
NetworkProtocol.SERVER_EVENTS = { EVENTS.CLIENT_READY, EVENTS.COMMAND }
NetworkProtocol.CLIENT_EVENTS = { EVENTS.RESPONSE, EVENTS.SNAPSHOT, EVENTS.DELTA, EVENTS.STATUS }
```

**信封结构**（所有命令/响应共用的 JSON 载荷）：

```json
{
  "version": 1,
  "requestId": "c-1700000000-1",
  "command": "Game.Strengthen",
  "payloadJson": "{\"slot\":\"weapon\"}"
}
```

## 3. 通用 Schema 校验器（核心）

`ValidateValue(value, schema, path)` 是递归校验器，支持：

| 规则字段 | 作用 |
|---------|------|
| `type` | `string/number/integer/boolean/table/array/object/any` |
| `enum` | 值必须在枚举内 |
| `min/max` | 数值范围 |
| `minLength/maxLength` | 字符串/数组长度 |
| `pattern` | 字符串正则（`string.match`） |
| `fields/properties` + `required` + `allowUnknown` | 对象字段表 |
| `items` | 数组元素 schema |
| `validate` | 自定义校验函数（`fn(value) → boolean, reason`） |

```lua
---@param value any
---@param schema table|nil
---@param path string|nil
---@return boolean
---@return table|nil          -- 失败时返回 ErrorCodes.Make 错误
function NetworkProtocol.ValidateValue(value, schema, path)
    if schema == nil then return true, nil end
    path = path or "$"

    if value == nil then
        if schema.required == true then
            return false, makeError(ErrorCodes.INVALID_PAYLOAD, "Required value is missing", path)
        end
        return true, nil
    end

    local expectedType = schema.type
    if expectedType ~= nil then
        local actualType = type(value)
        local typeMatches = false
        if expectedType == "any" then
            typeMatches = true
        elseif expectedType == "integer" then
            typeMatches = actualType == "number" and value == math.floor(value)
        elseif expectedType == "number" then
            typeMatches = actualType == "number" and value == value
        elseif expectedType == "object" or expectedType == "table" then
            typeMatches = actualType == "table"
        elseif expectedType == "array" then
            typeMatches = actualType == "table"
        else
            typeMatches = actualType == expectedType
        end
        if not typeMatches then
            return false, makeError(ErrorCodes.INVALID_PAYLOAD,
                "Expected " .. tostring(expectedType) .. ", got " .. actualType, path)
        end
    end

    -- enum / min / max / minLength / maxLength / pattern 逐一校验（略）
    ...

    if type(value) == "table" then
        local isArray = expectedType == "array"
        if isArray then
            local length = #value
            -- 数组长度与逐元素递归校验
            for index, item in ipairs(value) do
                local ok, err = NetworkProtocol.ValidateValue(item, schema.items, path .. "[" .. index .. "]")
                if not ok then return false, err end
            end
            -- 禁止稀疏数组（非连续 key）
            for key in pairs(value) do
                if type(key) ~= "number" or key < 1 or key > length or key ~= math.floor(key) then
                    return false, makeError(ErrorCodes.INVALID_PAYLOAD, "Array contains a non-sequential key", path)
                end
            end
        else
            local fields = schema.fields or schema.properties
            if fields ~= nil then
                -- 必填字段 + 逐字段递归校验
                for fieldName, fieldSchema in pairs(fields) do
                    if required[fieldName] or fieldSchema.required == true then
                        if value[fieldName] == nil then ... return false, err end
                    end
                    local ok, err = NetworkProtocol.ValidateValue(value[fieldName], fieldSchema, path .. "." .. fieldName)
                    if not ok then return false, err end
                end
                -- allowUnknown == false 时拒绝未声明字段
                if schema.allowUnknown == false then
                    for fieldName in pairs(value) do
                        if fields[fieldName] == nil then
                            return false, makeError(ErrorCodes.INVALID_PAYLOAD, "Unknown field", path .. "." .. tostring(fieldName))
                        end
                    end
                end
            end
        end
    end

    -- 自定义校验钩子
    if schema.validate ~= nil then
        local ok, valid, reason = pcall(schema.validate, value)
        if not ok then return false, makeError(ErrorCodes.INTERNAL_ERROR, "Schema validator failed", path) end
        if valid == false then return false, makeError(ErrorCodes.INVALID_PAYLOAD, reason or "Custom validation failed", path) end
    end
    return true, nil
end

--- 顶层入口：为 payload 套一层 "$payload" 根路径
function NetworkProtocol.ValidatePayload(payload, schema)
    if schema == nil then return true, nil end
    -- 兼容：只给了 fields 没给 type 时补 type="object"
    if schema.type == nil and (schema.fields ~= nil or schema.properties ~= nil) then
        local copy = {}
        for key, value in pairs(schema) do copy[key] = value end
        copy.type = "object"
        schema = copy
    end
    return NetworkProtocol.ValidateValue(payload, schema, "$payload")
end
```

## 4. 命令信封读取与校验

### 4.1 `ValidateRequest` —— 纯表校验（可单测）

```lua
function NetworkProtocol.ValidateRequest(request)
    if type(request) ~= "table" then return false, nil, makeError(ErrorCodes.INVALID_REQUEST) end

    local requestId = canonicalRequestField(request, "requestId", NetworkProtocol.FIELDS.REQUEST_ID)
    local command   = canonicalRequestField(request, "command",   NetworkProtocol.FIELDS.COMMAND)
    local payloadJson = canonicalRequestField(request, "payloadJson", NetworkProtocol.FIELDS.PAYLOAD_JSON)

    -- requestId：1~96 字符，字符白名单 ^[%w%._:%-]+$
    if type(requestId) ~= "string" or #requestId < 1 or #requestId > NetworkProtocol.MAX_REQUEST_ID_LENGTH
        or string.match(requestId, "^[%w%._:%-]+$") == nil then
        return false, nil, makeError(ErrorCodes.INVALID_REQUEST_ID)
    end

    -- command：字母开头，仅字母数字._:- ，最长 64
    if type(command) ~= "string" or #command < 1 or #command > NetworkProtocol.MAX_COMMAND_LENGTH
        or string.match(command, "^[%a][%w%._:%-]*$") == nil then
        return false, nil, makeError(ErrorCodes.INVALID_REQUEST, "Command is invalid")
    end

    -- payloadJson：可空（视为 "{}"），必须字符串，不得超过上限
    if payloadJson == nil or payloadJson == "" then payloadJson = "{}" end
    if type(payloadJson) ~= "string" then return false, nil, makeError(ErrorCodes.INVALID_REQUEST, "PayloadJson must be a string") end
    if #payloadJson > NetworkProtocol.MAX_PAYLOAD_BYTES then return false, nil, makeError(ErrorCodes.PAYLOAD_TOO_LARGE) end

    return true, { requestId = requestId, command = command, payloadJson = payloadJson }, nil
end
```

### 4.2 `ReadCommandEvent` —— 从 eventData 提取信封

从 UrhoX `VariantMap` 中兼容多种字段读取方式（`eventData[key]` 直接取 / `eventData:GetString(key)` / Variant `:GetString()`），全部 `pcall` 包裹，防御不同宿主差异。

```lua
function NetworkProtocol.ReadCommandEvent(eventData)
    return NetworkProtocol.ValidateRequest({
        requestId   = readRawField(eventData, NetworkProtocol.FIELDS.REQUEST_ID),
        command     = readRawField(eventData, NetworkProtocol.FIELDS.COMMAND),
        payloadJson = readRawField(eventData, NetworkProtocol.FIELDS.PAYLOAD_JSON),
    })
end
```

### 4.3 JSON 编解码与信封构造

```lua
-- 解码（限制大小，异常返回 INVALID_JSON）
function NetworkProtocol.DecodePayload(payloadJson, decoder)
    ...
    local ok, payload = pcall(decoder, payloadJson == "" and "{}" or payloadJson)
    if not ok or type(payload) ~= "table" then
        return false, nil, makeError(ErrorCodes.INVALID_JSON)
    end
    return true, payload, nil
end

-- 编码
function NetworkProtocol.EncodePayload(payload, encoder)
    ...
    local ok, encoded = pcall(encoder, payload == nil and {} or payload)
    if not ok or type(encoded) ~= "string" then
        return false, nil, makeError(ErrorCodes.INTERNAL_ERROR, "JSON encoding failed")
    end
    return true, encoded, nil
end

-- 组信封（version/requestId/command/payloadJson）
function NetworkProtocol.MakeEnvelope(requestId, command, payload, encoder)
    local ok, payloadJson, err = NetworkProtocol.EncodePayload(payload, encoder)
    if not ok then return nil, err end
    return { version = NetworkProtocol.VERSION, requestId = requestId, command = command, payloadJson = payloadJson }, nil
end

-- 信封 → VariantMap（发送用；无 VariantMap 环境时回退普通 table）
function NetworkProtocol.ToVariantMap(envelope)
    local values = {
        [NetworkProtocol.FIELDS.VERSION]     = envelope.version or NetworkProtocol.VERSION,
        [NetworkProtocol.FIELDS.REQUEST_ID]  = envelope.requestId or "",
        [NetworkProtocol.FIELDS.COMMAND]     = envelope.command or "",
        [NetworkProtocol.FIELDS.PAYLOAD_JSON]= envelope.payloadJson or "{}",
    }
    local variantMapConstructor, variantConstructor = _G["VariantMap"], _G["Variant"]
    if variantMapConstructor == nil or variantConstructor == nil then return values end
    local created, eventData = pcall(function() return variantMapConstructor() end)
    if not created or eventData == nil then return values end
    for fieldName, value in pairs(values) do eventData[fieldName] = variantConstructor(value) end
    return eventData
end
```

## 5. 远程事件注册

```lua
function NetworkProtocol.RegisterServerEvents(networkObject)
    networkObject = networkObject or _G["network"]
    assert(networkObject ~= nil, "Network subsystem is unavailable")
    for _, eventName in ipairs(NetworkProtocol.SERVER_EVENTS) do
        networkObject:RegisterRemoteEvent(eventName)
    end
end

function NetworkProtocol.RegisterClientEvents(networkObject)
    networkObject = networkObject or _G["network"]
    assert(networkObject ~= nil, "Network subsystem is unavailable")
    for _, eventName in ipairs(NetworkProtocol.CLIENT_EVENTS) do
        networkObject:RegisterRemoteEvent(eventName)
    end
end
```

> 调用点：服务端 `ServerMain.Start()` 调 `Protocol.RegisterServerEvents()`；客户端 `ServerClient.Start()` 调 `Protocol.RegisterClientEvents()`。

### 关键工厂函数

```lua
--- 描述：取错误的默认 reason
function ErrorCodes.Describe(code)
    return DESCRIPTIONS[code] or DESCRIPTIONS[ErrorCodes.INTERNAL_ERROR]
end

--- 构造错误对象
---@param code string
---@param reason string|nil
---@param details table|nil
---@return table
function ErrorCodes.Make(code, reason, details)
    local result = { ok = code == ErrorCodes.OK, code = code, reason = reason or ErrorCodes.Describe(code) }
    if details ~= nil then result.details = details end
    return result
end

--- 成功包装
function ErrorCodes.Success(data)
    return { ok = true, code = ErrorCodes.OK, data = data }
end

--- 判断是否为错误对象
function ErrorCodes.IsError(value)
    return type(value) == "table" and value.ok == false and type(value.code) == "string"
end
```

## 7. 使用模式小结

| 场景 | 调用 |
|------|------|
| 服务端解析命令事件 | `NetworkProtocol.ReadCommandEvent(eventData)` |
| 服务端校验 payload | `NetworkProtocol.ValidatePayload(payload, definition.schema)` |
| 服务端发响应 | `NetworkProtocol.MakeEnvelope(...)` + `ToVariantMap` + `connection:SendRemoteEvent` |
| 客户端发请求 | `NetworkProtocol.MakeCommandEvent(command, requestId, payload, encoder)` |
| 客户端解析快照/增量 | `NetworkProtocol.DecodePayload(eventData[PAYLOAD_JSON]:GetString())` |

> ⚠️ 注意：`CommandRouter.Dispatch` 会**重复做** `ReadCommandEvent` + `DecodePayload` + `ValidatePayload`。`ValidateRequest` 只校验信封格式；`ValidatePayload` 用命令定义里的 `definition.schema` 校验业务字段。两层校验职责不同，缺一不可。
